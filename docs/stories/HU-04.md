## Modelo de datos

### Migración Prisma

```prisma
model Food {
  // ... campos existentes ...
  verifiedAt   DateTime?  // null = comunidad, fecha = verificado por admin
  verifiedBy   String?    // email admin que verificó
  needsReview  Boolean    @default(false)
  voteScore    Int        @default(0)  // denormalized: sum of votes
  voteCount    Int        @default(0)  // denormalized: total votes cast

  votes        FoodVote[]
}

model FoodVote {
  id        Int      @id @default(autoincrement())
  userId    Int
  foodId    Int
  vote      Int      // +1 o -1
  createdAt DateTime @default(now())

  user      User     @relation(fields: [userId], references: [id], onDelete: Cascade)
  food      Food     @relation(fields: [foodId], references: [id], onDelete: Cascade)

  @@unique([userId, foodId])  // un voto por usuario por alimento
  @@index([foodId])
}

model User {
  // ... campos existentes ...
  votes     FoodVote[]
}
```

### Migración Turso producción

```sql
ALTER TABLE Food ADD COLUMN verifiedAt DATETIME;
ALTER TABLE Food ADD COLUMN verifiedBy TEXT;
ALTER TABLE Food ADD COLUMN needsReview INTEGER DEFAULT 0;
ALTER TABLE Food ADD COLUMN voteScore INTEGER DEFAULT 0;
ALTER TABLE Food ADD COLUMN voteCount INTEGER DEFAULT 0;

CREATE TABLE FoodVote (
  id INTEGER PRIMARY KEY AUTOINCREMENT,
  userId INTEGER NOT NULL,
  foodId INTEGER NOT NULL,
  vote INTEGER NOT NULL,
  createdAt DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(userId, foodId)
);
CREATE INDEX FoodVote_foodId_idx ON FoodVote(foodId);

-- Marcar todos los alimentos USDA existentes como verificados
UPDATE Food SET verifiedAt = CURRENT_TIMESTAMP, verifiedBy = 'system'
WHERE source = 'usda';
```

## Endpoints nuevos

### `POST /api/foods/[id]/vote`
**Body:**
```json
{ "vote": 1 }   // o -1
```

**Response:**
```json
{ "ok": true, "voteScore": 5, "voteCount": 7, "userVote": 1 }
```

**Lógica:**
- Upsert en `FoodVote` (un voto por usuario)
- Recalcular `voteScore` y `voteCount` denormalizados
- Si el cambio resulta en `voteScore < -3` y el food no está verificado → marcar `needsReview = true`

### `DELETE /api/foods/[id]/vote`
Retracta el voto del usuario.

### `GET /api/admin/foods/review` (admin only)
**Auth check:** session.user.email en allowlist de admins (env var `ADMIN_EMAILS`)
**Response:**
```json
{
  "foods": [
    { "id": 12, "nombre": "Papa colombiana", "cal": 500, "voteScore": -5, "categoria": "carbohidrato", "needsReview": true }
  ]
}
```

### `POST /api/admin/foods/[id]/verify` (admin only)
Marca `verifiedAt = NOW()`, `verifiedBy = admin email`, `needsReview = false`.

### `app/components/FoodVoteButton.tsx`

```tsx
"use client";
import { useState } from "react";
import { ChevronUp, ChevronDown } from "lucide-react";

type Props = {
  foodId: number;
  initialScore: number;
  initialUserVote: -1 | 0 | 1;
};

export default function FoodVoteButton({ foodId, initialScore, initialUserVote }: Props) {
  const [score, setScore] = useState(initialScore);
  const [vote, setVote] = useState(initialUserVote);

  async function castVote(direction: 1 | -1) {
    const newVote = vote === direction ? 0 : direction;
    const delta = newVote - vote;
    // Optimistic update
    setScore(score + delta);
    setVote(newVote);

    if (newVote === 0) {
      await fetch(`/api/foods/${foodId}/vote`, { method: "DELETE" });
    } else {
      await fetch(`/api/foods/${foodId}/vote`, {
        method: "POST",
        body: JSON.stringify({ vote: newVote }),
      });
    }
  }

  return (
    <div className="flex items-center gap-1">
      <button onClick={() => castVote(1)} className={vote === 1 ? "text-macro-protein" : "text-text-tertiary"}>
        <ChevronUp size={14} />
      </button>
      <span className="text-xs tabular-nums w-6 text-center">{score}</span>
      <button onClick={() => castVote(-1)} className={vote === -1 ? "text-accent-warm" : "text-text-tertiary"}>
        <ChevronDown size={14} />
      </button>
    </div>
  );
}
```

## Algoritmo de detección de outliers

### `lib/foods/outlier-detection.ts`

```ts
/**
 * Para cada categoría (proteina, carbo, grasa, verdura, fruta, lacteo, otros),
 * calcula media y desviación estándar de calorías por 100g.
 * Marca como needsReview cualquier alimento que esté a más de 3σ de la media.
 */
export async function detectOutliers() {
  const categories = await prisma.food.findMany({
    distinct: ["categoria"],
    select: { categoria: true },
  });

  for (const { categoria } of categories) {
    const foods = await prisma.food.findMany({
      where: { categoria, verifiedAt: null },
    });
    if (foods.length < 10) continue; // muy pocos para estadística

    const cals = foods.map((f) => f.cal);
    const mean = cals.reduce((s, c) => s + c, 0) / cals.length;
    const variance = cals.reduce((s, c) => s + (c - mean) ** 2, 0) / cals.length;
    const stddev = Math.sqrt(variance);

    const outliers = foods.filter((f) => Math.abs(f.cal - mean) > 3 * stddev);

    for (const food of outliers) {
      await prisma.food.update({
        where: { id: food.id },
        data: { needsReview: true },
      });
    }

    console.log(`[outliers ${categoria}] ${outliers.length} marked, mean=${mean.toFixed(1)}, σ=${stddev.toFixed(1)}`);
  }
}
```

Ejecutar via cron Vercel: `vercel.json` → `crons: [{ path: "/api/cron/outliers", schedule: "0 4 * * 1" }]` (lunes 4am).

import { detectOutliers } from "@/lib/foods/outlier-detection";

it("marks foods >3σ from category mean", async () => {
  // Setup: 20 alimentos categoria "proteina" con media ~120 kcal, σ ~20
  // + 1 alimento con cal=500 (outlier obvio)
  await detectOutliers();
  const outlier = await prisma.food.findFirst({ where: { cal: 500 } });
  expect(outlier?.needsReview).toBe(true);
});

it("ignores categories with <10 items", async () => {
  // Solo 5 alimentos de "experimental" — no debe marcar nada
  await detectOutliers();
  const experimental = await prisma.food.findMany({ where: { categoria: "experimental" } });
  expect(experimental.every((f) => !f.needsReview)).toBe(true);
});

describe("POST /api/foods/[id]/vote", () => {
  it("upserts vote and updates denormalized score", async () => {
    const res = await POST(mockRequest({ vote: 1 }), { id: 1 });

    expect(data.voteScore).toBe(-1);

    expect(food?.needsReview).toBe(true);
  });
});

## Riesgos específicos

| Riesgo | Mitigación |
|---|---|
| Votos negativos pueden ser usados maliciosamente | Limitar a 1 voto por usuario por food (unique constraint) |
| Outliers falsos positivos (alimentos válidos pero extremos) | Admin puede marcar `verifiedAt` manualmente, lo que limpia `needsReview` |
| Migración rompe queries existentes que no esperan voteScore | Defaults `0` evitan NULL; tests de regresión cubren |
| UI con voto puede saturar `StepSearch` visualmente | Mostrar voto solo en detalle, no en lista |
