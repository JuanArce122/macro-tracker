## Endpoints nuevos

### `GET /api/foods/barcode/[code]`

**Flujo:**
1. Buscar en DB local: `prisma.food.findUnique({ where: { barcode: code } })`
2. Si existe → retornar con header `x-cache: hit`
3. Si no → fetch a OFF: `https://world.openfoodfacts.org/api/v0/product/{code}.json`
4. Si OFF retorna producto válido → guardar en DB y retornar
5. Si OFF retorna 404 → retornar 404 con sugerencia "Crear manualmente"
6. Si el usuario tiene `countryCode`, priorizar variantes regionales:
   - OFF retorna múltiples versiones según país → matchear `countries_tags` contra `user.countryCode`

**Response (200 — existe):**
```json
{
  "food": {
    "id": 245,
    "barcode": "7702049000123",
    "nombre": "Coca-Cola 250ml",
    "cal": 100,
    "p": 0,
    "c": 27,
    "f": 0,
    "source": "openfoodfacts",
    "regionCode": "CO"
  },
  "cached": true
}
```

**Response (404 — no existe):**
```json
{
  "error": "Producto no encontrado",
  "suggestion": "create_manual",
  "barcode": "9999999999999"
}
```

## Componente UI: `StepBarcode.tsx`

```tsx
"use client";

import { useEffect, useRef, useState } from "react";
import { BrowserMultiFormatReader } from "@zxing/browser";
import { Camera, X, AlertTriangle, Loader2 } from "lucide-react";
import Icon from "@/app/components/ui/Icon";
import Button from "@/app/components/ui/Button";

type Props = {
  onResult: (food: Food) => void;
  onNotFound: (barcode: string) => void;
  onBack: () => void;
};

export default function StepBarcode({ onResult, onNotFound, onBack }: Props) {
  const videoRef = useRef<HTMLVideoElement>(null);
  const [scanning, setScanning] = useState(true);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    const reader = new BrowserMultiFormatReader();
    let stopped = false;

    reader.decodeFromVideoDevice(null, videoRef.current!, async (result, err) => {
      if (stopped) return;
      if (result) {
        stopped = true;
        setScanning(false);
        setLoading(true);
        const barcode = result.getText();
        const res = await fetch(`/api/foods/barcode/${barcode}`);
        if (res.ok) {
          const { food } = await res.json();
          onResult(food);
        } else {
          onNotFound(barcode);
        }
      }
    }).catch(() => setError("No se pudo acceder a la cámara"));

    return () => {
      stopped = true;
      reader.reset();
    };
  }, []);

  return (
    <div className="flex flex-col flex-1 bg-bg-primary">
      <button onClick={onBack} className="absolute top-4 left-4 z-10 bg-black/40 text-white rounded-full p-2">
        <Icon icon={X} size={20} />
      </button>

      <div className="relative flex-1 overflow-hidden">
        <video ref={videoRef} className="w-full h-full object-cover" />
        {/* Marco de escaneo */}
        <div className="absolute inset-0 flex items-center justify-center pointer-events-none">
          <div className="w-3/4 h-32 border-2 border-white/80 rounded-xl" />
        </div>
        {loading && (
          <div className="absolute inset-0 bg-black/40 flex items-center justify-center">
            <Icon icon={Loader2} size={32} className="text-white animate-spin" />
          </div>
        )}
      </div>

      <div className="p-5 text-center bg-bg-primary">
        <p className="text-text-tertiary text-sm">
          {scanning ? "Apunta al código de barras" : "Detectado, buscando…"}
        </p>
        {error && (
          <div className="mt-3 flex items-center gap-2 text-accent-warm text-sm justify-center">
            <Icon icon={AlertTriangle} size={16} />
            {error}
          </div>
        )}
      </div>
    </div>
  );
}
```

## Lib helper: `lib/barcode.ts`

```ts
const OFF_BASE = "https://world.openfoodfacts.org/api/v0/product";

/**
 * Valida que el barcode tenga formato EAN-13, EAN-8, UPC-A o UPC-E.
 */
export function isValidBarcode(code: string): boolean {
  if (!/^\d{8,13}$/.test(code)) return false;
  // Checksum básico (Luhn-like) opcional
  return true;
}

export type OFFProductDetailed = {
  product?: {
    product_name?: string;
    brands?: string;
    nutriments?: {
      "energy-kcal_100g"?: number;
      proteins_100g?: number;
      carbohydrates_100g?: number;
      fat_100g?: number;
    };
    countries_tags?: string[]; // ej. ["en:colombia", "en:mexico"]
  };
  status: 0 | 1;
};

export async function fetchFromOFF(barcode: string): Promise<OFFProductDetailed | null> {
  const controller = new AbortController();
  const timeout = setTimeout(() => controller.abort(), 4000);

  try {
    const res = await fetch(`${OFF_BASE}/${barcode}.json`, { signal: controller.signal });
    if (!res.ok) return null;
    return res.json();
  } catch {
    return null;
  } finally {
    clearTimeout(timeout);
  }
}

export function matchesRegion(countriesTags: string[] | undefined, countryCode: string | null): boolean {
  if (!countryCode || !countriesTags) return false;
  return countriesTags.some((t) => t.toLowerCase().includes(countryCode.toLowerCase()));
}
```

describe("isValidBarcode", () => {
  it("accepts EAN-13", () => {
    expect(isValidBarcode("7702049000123")).toBe(true);
  });
  it("rejects letters", () => {
    expect(isValidBarcode("ABC123")).toBe(false);

  it("rejects too short", () => {
    expect(isValidBarcode("123")).toBe(false);
  });

  it("matches Colombia tag", () => {
    expect(matchesRegion(["en:colombia"], "CO")).toBe(true);
  });
  it("returns false when no countryCode", () => {
    expect(matchesRegion(["en:colombia"], null)).toBe(false);
  });
});

    expect(res.status).toBe(200);
