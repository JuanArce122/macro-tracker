## Algoritmo

### `lib/weight-trend.ts`

```ts
type WeightPoint = { date: string; weightKg: number };

/**
 * Smoothing exponencial simple sobre la serie temporal.
 * alpha = 0.3 (más peso a observaciones recientes pero con suavizado)
 */
export function smoothExponential(points: WeightPoint[], alpha = 0.3): number[] {
  if (points.length === 0) return [];
  const smoothed = [points[0].weightKg];
  for (let i = 1; i < points.length; i++) {
    smoothed.push(alpha * points[i].weightKg + (1 - alpha) * smoothed[i - 1]);
  }
  return smoothed;
}

/**
 * Pendiente (kg/semana) calculada por regresión lineal sobre los puntos suavizados.
 */
export function calcTrendSlope(points: WeightPoint[]): number {
  if (points.length < 7) return 0; // necesitamos al menos 1 semana

  const smoothed = smoothExponential(points);
  const n = smoothed.length;
  const xs = smoothed.map((_, i) => i);
  const ys = smoothed;
  const meanX = xs.reduce((s, x) => s + x, 0) / n;
  const meanY = ys.reduce((s, y) => s + y, 0) / n;
  const num = xs.reduce((s, x, i) => s + (x - meanX) * (ys[i] - meanY), 0);
  const den = xs.reduce((s, x) => s + (x - meanX) ** 2, 0);
  const slopePerDay = num / den;
  return slopePerDay * 7; // kg/semana
}

/**
 * Calcula adherencia: % de días con al menos 1 comida en los últimos 14 días.
 */
export async function calcAdherence(userId: number): Promise<number> {
  const fourteenDaysAgo = new Date();
  fourteenDaysAgo.setDate(fourteenDaysAgo.getDate() - 14);
  const fourteenAgoStr = fourteenDaysAgo.toISOString().split("T")[0];

  const distinctDays = await prisma.meal.groupBy({
    by: ["dateLocal"],
    where: { userId, dateLocal: { gte: fourteenAgoStr } },
  });
  return distinctDays.length / 14;
}

/**
 * Decide qué ajuste de meta proponer.
 * - Si fitnessGoal = "lose": target -0.5 kg/sem. Si tendencia > -0.2 (perdiendo lento), bajar calorías -100. Si < -1.0 (perdiendo rápido), subir +100.
 * - Si fitnessGoal = "gain": target +0.25 kg/sem. Inverso.
 * - Si fitnessGoal = "maintain": target 0 kg/sem.
 */
export function decideAdjustment(
  currentGoal: { calories: number; protein: number; carbs: number; fat: number },
  trendKgPerWeek: number,
  fitnessGoal: "lose" | "maintain" | "gain"
): { newGoal: typeof currentGoal; reason: string } | null {
  const TARGETS: Record<string, number> = { lose: -0.5, maintain: 0, gain: 0.25 };
  const target = TARGETS[fitnessGoal];
  const delta = trendKgPerWeek - target;

  // Solo ajusta si está fuera de ±0.2 kg/sem del target
  if (Math.abs(delta) < 0.2) return null;

  // Cada 100 kcal/día ≈ 0.1 kg/sem (regla aproximada)
  const calorieAdjust = Math.round(-delta * 1000) * 10; // redondeo a múltiplos de 10
  const clampedAdjust = Math.max(-200, Math.min(200, calorieAdjust));

  const newCalories = currentGoal.calories + clampedAdjust;
  // Mantener proporciones P/C/G
  const ratio = newCalories / currentGoal.calories;
  return {
    newGoal: {
      calories: newCalories,
      protein: Math.round(currentGoal.protein * ratio),
      carbs: Math.round(currentGoal.carbs * ratio),
      fat: Math.round(currentGoal.fat * ratio),
    },
    reason: delta > 0
      ? (fitnessGoal === "lose" ? "trend_loss_too_slow" : "trend_gain_too_fast")
      : (fitnessGoal === "lose" ? "trend_loss_too_fast" : "trend_gain_too_slow"),
  };
}
```

## Endpoints

### `POST /api/weight`
**Body:** `{ "weightKg": 70.5, "date": "2026-05-22" }` (date opcional, default hoy)
**Response:** `{ ok: true, entry: { id, date, weightKg } }`

### `GET /api/weight?limit=30`
**Response:**
```json
{
  "entries": [{ "date": "2026-05-22", "weightKg": 70.5 }],
  "trendKgPerWeek": -0.3,
  "adherence": 0.85
}
```

### `POST /api/goals/suggest`
Calcula ajuste sin aplicar. Solo si han pasado ≥14 días desde el último cambio Y hay ≥7 weight entries.
**Response:**
```json
{
  "shouldSuggest": true,
  "currentGoal": { "calories": 2000, "protein": 150, "carbs": 200, "fat": 65 },
  "suggestedGoal": { "calories": 1900, "protein": 145, "carbs": 190, "fat": 62 },
  "reason": "trend_loss_too_slow",
  "trendKgPerWeek": -0.2,
  "explanation": "Tu peso ha bajado más lento que tu objetivo. Sugerimos reducir 100 kcal/día."
}
```

### `POST /api/goals/apply-suggestion`
Aplica la última sugerencia, registra en `GoalAdjustmentLog`.

## Tests

### Unit

**`__tests__/lib/weight-trend.test.ts`**
```ts
import { smoothExponential, calcTrendSlope, decideAdjustment } from "@/lib/weight-trend";

describe("smoothExponential", () => {
  it("returns single value for single point", () => {
    expect(smoothExponential([{ date: "1", weightKg: 70 }])).toEqual([70]);
  });

  it("smooths upward trend", () => {
    const points = [70, 71, 72, 73, 74].map((w, i) => ({ date: `d${i}`, weightKg: w }));
    const smoothed = smoothExponential(points, 0.3);
    expect(smoothed[smoothed.length - 1]).toBeLessThan(74); // suavizado, no llega al pico
  });
});

describe("calcTrendSlope", () => {
  it("returns 0 with insufficient data", () => {
    expect(calcTrendSlope([])).toBe(0);
    expect(calcTrendSlope([{ date: "1", weightKg: 70 }])).toBe(0);
  });

  it("detects negative trend (losing weight)", () => {
    const points = Array.from({ length: 14 }, (_, i) => ({
      date: `d${i}`, weightKg: 75 - i * 0.05,
    }));
    expect(calcTrendSlope(points)).toBeLessThan(0);
  });
});

describe("decideAdjustment", () => {
  it("suggests reducing calories when losing too slow", () => {
    const result = decideAdjustment(
      { calories: 2000, protein: 150, carbs: 200, fat: 65 },
      -0.1,  // perdiendo 0.1 kg/sem
      "lose" // pero quiere perder
    );
    expect(result?.newGoal.calories).toBeLessThan(2000);
    expect(result?.reason).toBe("trend_loss_too_slow");
  });

  it("returns null when trend is within target ±0.2", () => {
    const result = decideAdjustment(
      { calories: 2000, protein: 150, carbs: 200, fat: 65 },
      -0.45, // muy cerca de -0.5 target
      "lose"
    );
    expect(result).toBeNull();
  });
});
```

## Riesgos específicos

| Riesgo | Mitigación |
|---|---|
| Algoritmo MAPE alto en usuarios con peso muy fluctuante | Cap el ajuste a ±200 kcal/cambio; intervalo mínimo 14 días |
| Modo Auto puede frustrar usuarios sin avisar | Notificación push obligatoria + log auditable |
| Suavizado puede ocultar cambios reales | Mostrar ambos: línea raw + smoothed |
| Sin pesar por días → trend distorsionado | Excluir gaps > 5 días del cálculo |
