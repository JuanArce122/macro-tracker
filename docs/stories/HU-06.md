Generar planes de comidas semanales según las metas, preferencias dietéticas, alergias, presupuesto y tiempo de cocina. Incluir lista de compras consolidada y registro automático en el diario.

```prisma
model Recipe {
  id            Int       @id @default(autoincrement())
  name          String
  description   String?
  instructions  String    // texto plano o markdown
  prepTimeMin   Int
  cookTimeMin   Int
  servings      Int       @default(1)
  // Tags
  dietTags      String    // CSV: "vegano,vegetariano,keto,paleo,sin-gluten"
  allergyTags   String    // CSV: "lacteos,gluten,frutos-secos,mariscos,huevo,soja"
  costLevel     Int       // 1-3 (bajo, medio, alto)
  cuisineCode   String?   // "CO", "MX", "ITA", "JPN", etc.
  source        String    @default("seed")  // "seed" | "user" | "imported"
  userId        Int?      // null = receta pública
  imageUrl      String?
  createdAt     DateTime  @default(now())

  ingredients   RecipeIngredient[]
  plannedMeals  PlannedMeal[]

  @@index([cuisineCode])
}

model RecipeIngredient {
  id              Int     @id @default(autoincrement())
  recipeId        Int
  foodId          Int
  gramsPerServing Float
  optional        Boolean @default(false)

  recipe          Recipe  @relation(fields: [recipeId], references: [id], onDelete: Cascade)
  food            Food    @relation(fields: [foodId], references: [id])

  @@index([recipeId])
}

model MealPlan {
  id          Int       @id @default(autoincrement())
  userId      Int
  user        User      @relation(fields: [userId], references: [id], onDelete: Cascade)
  startDate   String    // YYYY-MM-DD (lunes)
  endDate     String
  preferences String    // JSON: { diet, allergies, mealsPerDay, maxPrepMinutes, costLevel }
  isActive    Boolean   @default(true)
  createdAt   DateTime  @default(now())

  plannedMeals PlannedMeal[]

  @@index([userId, startDate])
}

model PlannedMeal {
  id          Int      @id @default(autoincrement())
  planId      Int
  date        String   // YYYY-MM-DD
  mealType    String   // "breakfast" | "lunch" | "dinner" | "snack"
  recipeId    Int
  servings    Float    @default(1)
  consumed    Boolean  @default(false)  // marcado cuando se registra como Meal
  consumedAt  DateTime?

  plan        MealPlan @relation(fields: [planId], references: [id], onDelete: Cascade)
  recipe      Recipe   @relation(fields: [recipeId], references: [id])

  @@unique([planId, date, mealType])
  @@index([date])
}

model Food {
  // ... existentes ...
  recipeIngredients RecipeIngredient[]
}
```

## Motor de planificación (`lib/meal-planner/generator.ts`)

```ts
type Preferences = {
  diet: ("vegano" | "vegetariano" | "keto" | "paleo" | "sin-gluten")[];
  allergies: string[];
  mealsPerDay: 3 | 4 | 5;
  maxPrepMinutes: number;
  costLevel: 1 | 2 | 3;
};

type Goal = { calories: number; protein: number; carbs: number; fat: number };

/**
 * Algoritmo greedy:
 * 1. Filtra recetas por dieta/alergia/tiempo/costo
 * 2. Para cada día y cada slot, escoge la receta que mejor cumpla macros del slot
 *    sin repetir más de 1 vez en la semana
 * 3. Si no encuentra match, ajusta servings para acercarse
 * 4. Variedad por cuisineCode (max 3 del mismo país por semana)
 */
export async function generatePlan(
  userId: number,
  goal: Goal,
  preferences: Preferences,
  startDate: string
): Promise<PlannedMeal[]> {
  const candidates = await prisma.recipe.findMany({
    where: {
      // Filtros por preferences
      AND: [
        // Diet inclusive
        ...preferences.diet.map((d) => ({ dietTags: { contains: d } })),
        // Alergias exclusive
        ...preferences.allergies.map((a) => ({ NOT: { allergyTags: { contains: a } } })),
        { prepTimeMin: { lte: preferences.maxPrepMinutes } },
        { costLevel: { lte: preferences.costLevel } },
      ],
      OR: [{ userId: null }, { userId }],
    },
    include: { ingredients: { include: { food: true } } },
  });

  if (candidates.length < 14) {
    throw new Error("No hay suficientes recetas que cumplan tus preferencias. Relaja restricciones.");
  }

  // Distribución macros por slot
  const slotRatio: Record<string, number> = {
    breakfast: 0.25, lunch: 0.35, dinner: 0.30, snack: 0.10,
  };
  const slots: string[] = preferences.mealsPerDay === 5 ? ["breakfast","snack","lunch","snack","dinner"]
                       : preferences.mealsPerDay === 4 ? ["breakfast","lunch","snack","dinner"]
                       : ["breakfast","lunch","dinner"];

  const plan: Array<{ date: string; mealType: string; recipeId: number; servings: number }> = [];
  const usedRecipes = new Map<number, number>(); // recipeId → count
  const countriesUsed = new Map<string, number>();

  for (let dayOffset = 0; dayOffset < 7; dayOffset++) {
    const date = addDays(startDate, dayOffset);
    for (const slot of slots) {
      const targetCal = goal.calories * (slotRatio[slot] ?? 0.1);
      const best = chooseBestRecipe(candidates, targetCal, usedRecipes, countriesUsed);
      const servings = scaleServings(best, targetCal);
      plan.push({ date, mealType: slot, recipeId: best.id, servings });
      usedRecipes.set(best.id, (usedRecipes.get(best.id) ?? 0) + 1);
      if (best.cuisineCode) countriesUsed.set(best.cuisineCode, (countriesUsed.get(best.cuisineCode) ?? 0) + 1);
    }
  }
  return plan;
}
```

## Endpoints

### `POST /api/meal-plans`
**Body:** `{ preferences: {...}, startDate: "2026-05-25" }`
**Response:** `{ planId: 45, plannedMeals: [...] }`
- Si user tiene plan activo → desactiva el anterior
- Genera vía `generatePlan`

### `GET /api/meal-plans/active`
**Response:** plan activo del usuario con `plannedMeals` y `recipes` populated

### `POST /api/meal-plans/[id]/log-meal`
**Body:** `{ plannedMealId: 123, weightG?: ... }`
- Crea un `Meal` real basado en la receta + servings
- Marca `PlannedMeal.consumed = true`, `consumedAt = now`

### `GET /api/meal-plans/[id]/shopping-list` (Sprint 12)
Consolida ingredientes:
```json
{
  "items": [
    { "foodId": 1, "nombre": "Arroz", "totalGrams": 1400, "recipes": ["Arroz con pollo", "Bandeja paisa"] },
    { "foodId": 2, "nombre": "Pollo", "totalGrams": 800, "recipes": ["Arroz con pollo", "Sancocho"] }
  ]
}
```

### `POST /api/meal-plans/[id]/replace-meal` (Sprint 12)
**Body:** `{ plannedMealId, reason: "no_me_gusta" | "no_tengo_ingrediente" }`
- Encuentra receta alternativa con macros equivalentes (±10%)
- Reemplaza en `PlannedMeal`

### `POST /api/recipes/import` (Sprint 12)
**Body:** `{ url: "https://example.com/receta" }`
- Fetch HTML
- Parse JSON-LD `<script type="application/ld+json">`
- Si `@type === "Recipe"` → extraer name, recipeIngredient[], cookTime, etc.
- Matchear ingredientes contra DB Food (HU-04 helps)
- Crear Recipe con `source = "imported"`, `userId`

## Tests

### Unit

**`__tests__/lib/meal-planner.test.ts`**
```ts
describe("generatePlan", () => {
  it("respects allergy restrictions", async () => {
    const plan = await generatePlan(userId, goal, { ...prefs, allergies: ["lacteos"] }, "2026-05-25");
    for (const meal of plan) {
      const recipe = await prisma.recipe.findUnique({ where: { id: meal.recipeId } });
      expect(recipe?.allergyTags).not.toContain("lacteos");
    }
  });

  it("doesn't repeat a recipe more than 2x per week", async () => {
    const plan = await generatePlan(userId, goal, prefs, "2026-05-25");
    const counts = plan.reduce((acc, m) => ({ ...acc, [m.recipeId]: (acc[m.recipeId] ?? 0) + 1 }), {} as Record<number, number>);
    expect(Math.max(...Object.values(counts))).toBeLessThanOrEqual(2);
  });

  it("throws when no recipes match preferences", async () => {
    await expect(generatePlan(userId, goal, { diet: ["vegano", "keto"], allergies: ["frutos-secos","lacteos","gluten","mariscos","huevo","soja"] }, "2026-05-25")).rejects.toThrow();
  });
});
```

## Riesgos específicos

| Riesgo | Mitigación |
|---|---|
| 50 recetas insuficientes para variedad | Iteración: añadir 100+ post-launch |
| Motor greedy puede no converger en macros exactos | Permitir ±15% tolerancia; mostrar diff al usuario |
| Recetas con ingredientes no en DB Food | Match aproximado + crear `Food` placeholder con `source = "recipe_ingredient"` |
| Generación lenta (>5s) | Cache plans por preferences hash; pre-computar |
| Importación URL JSON-LD falla en sitios sin schema.org | Fallback: pedir al usuario llenar manualmente |
| Lista compras agrupa unidades diferentes (1 huevo + 50g huevo) | Normalizar a gramos siempre |
