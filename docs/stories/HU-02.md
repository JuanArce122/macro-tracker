## Objetivo
Permitir al usuario dictar lo que comió en español colombiano y que la app parsee cantidades, alimentos y los matchee contra la base de datos.

## Componente: `StepVoice.tsx`

```tsx
"use client";
import { useEffect, useRef, useState } from "react";
import { Mic, MicOff, ChevronLeft, AlertTriangle, Loader2 } from "lucide-react";
import Button from "@/app/components/ui/Button";
import Icon from "@/app/components/ui/Icon";

type Props = {
  onResult: (items: ParsedItem[]) => void;
  onBack: () => void;
};

export default function StepVoice({ onResult, onBack }: Props) {
  const recognitionRef = useRef<SpeechRecognition | null>(null);
  const [recording, setRecording] = useState(false);
  const [transcript, setTranscript] = useState("");
  const [interimTranscript, setInterimTranscript] = useState("");
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    const SR = (window as any).SpeechRecognition || (window as any).webkitSpeechRecognition;
    if (!SR) {
      setError("Tu navegador no soporta dictado por voz");
      return;
    }
    const recognition = new SR();
    recognition.lang = "es-CO";
    recognition.continuous = true;
    recognition.interimResults = true;

    recognition.onresult = (event: SpeechRecognitionEvent) => {
      let finalText = "";
      let interimText = "";
      for (let i = event.resultIndex; i < event.results.length; i++) {
        const result = event.results[i];
        if (result.isFinal) finalText += result[0].transcript;
        else interimText += result[0].transcript;
      }
      if (finalText) setTranscript((t) => t + finalText);
      setInterimTranscript(interimText);
    };

    recognition.onerror = () => setError("Error al escuchar. Intenta de nuevo.");
    recognition.onend = () => setRecording(false);

    recognitionRef.current = recognition;
    return () => recognition.stop();
  }, []);

  function startRecording() {
    setTranscript("");
    setInterimTranscript("");
    setError(null);
    recognitionRef.current?.start();
    setRecording(true);
  }

  function stopRecording() {
    recognitionRef.current?.stop();
    setRecording(false);
  }

  async function handleProcess() {
    if (!transcript.trim()) return;
    setLoading(true);
    setError(null);
    try {
      const res = await fetch("/api/parse-voice", {
        method: "POST",
        headers: { "Content-Type": "application/json" },
        body: JSON.stringify({ transcript }),
      });
      if (!res.ok) {
        setError("No se pudo procesar el audio. Intenta más despacio o usa la búsqueda manual.");
        return;
      }
      const { items } = await res.json();
      onResult(items);
    } finally {
      setLoading(false);
    }
  }

  return (
    <div className="flex flex-col flex-1 p-5 bg-bg-primary">
      <button onClick={onBack} className="flex items-center gap-1 text-text-tertiary text-sm mb-6">
        <Icon icon={ChevronLeft} size={16} />
        Atrás
      </button>

      <h1 className="font-serif text-3xl tracking-[-0.02em] text-text-primary mb-8">Registro por voz</h1>

      {error && (
        <div className="bg-bg-secondary border border-border rounded-xl p-4 mb-4 flex items-start gap-2.5">
          <Icon icon={AlertTriangle} size={18} className="text-accent-warm mt-0.5" />
          <p className="text-text-primary text-sm">{error}</p>
        </div>
      )}

      <div className="bg-bg-secondary border border-border rounded-xl p-5 min-h-32 mb-6">
        <p className="text-text-primary text-base leading-relaxed">
          {transcript}
          <span className="text-text-tertiary">{interimTranscript}</span>
        </p>
        {!transcript && !interimTranscript && (
          <p className="text-text-tertiary text-sm italic">
            Ejemplo: "dos huevos revueltos con una arepa y café con leche"
          </p>
        )}
      </div>

      <button
        onClick={recording ? stopRecording : startRecording}
        disabled={!!error}
        className={`w-24 h-24 mx-auto rounded-full flex items-center justify-center transition-colors ${
          recording ? "bg-accent-warm" : "bg-text-primary"
        } text-bg-primary`}
      >
        <Icon icon={recording ? MicOff : Mic} size={32} />
      </button>
      <p className="text-center text-xs uppercase tracking-[0.08em] text-text-tertiary mt-3">
        {recording ? "Toca para detener" : "Toca para hablar"}
      </p>

      <Button
        variant="primary"
        size="lg"
        onClick={handleProcess}
        disabled={!transcript.trim() || loading}
        leadingIcon={loading ? <Icon icon={Loader2} size={18} className="animate-spin" /> : undefined}
        className="w-full mt-auto"
      >
        {loading ? "Procesando…" : "Procesar dictado"}
      </Button>
    </div>
  );
}
```

```ts
export const VOICE_PARSE_PROMPT = `Eres un nutricionista colombiano que parsea dictados informales sobre comida.
Te llega un transcript en español de Colombia. Identifica TODOS los alimentos mencionados con su cantidad.

Reglas:
- Cantidades implícitas: "un huevo" = 1 unidad, "una taza" = 240ml, "un plato" = porción estándar regional, "un poco" = porción pequeña sugerida
- Modismos colombianos: "bolillo" = pan blanco, "arepa" = arepa de maíz, "buñuelo" = buñuelo navideño, "patacones" = plátano frito, "ajiaco" = sopa colombiana
- Si la cantidad es ambigua, usa una porción estándar y marca confianza menor
- Ignora muletillas: "este", "como", "eh", "o sea"

Responde ÚNICAMENTE con JSON, sin markdown, sin explicaciones:
{
  "items": [
    {
      "nombre": "nombre del alimento normalizado",
      "cantidad": número,
      "unidad": "g" | "ml" | "unidad" | "taza" | "cucharada" | "porcion",
      "confianza": 0-1
    }
  ]
}

Si no se identifica ningún alimento, retorna { "items": [] }.`;
```

### `app/api/parse-voice/route.ts`

```ts
import { GoogleGenerativeAI } from "@google/generative-ai";
import { auth } from "@/auth";
import { VOICE_PARSE_PROMPT } from "@/lib/voice/parse-prompt";
import { matchFoods } from "@/lib/voice/match-foods";

const genAI = new GoogleGenerativeAI(process.env.GEMINI_API_KEY!);

export async function POST(req: Request) {
  const session = await auth();
  if (!session?.user?.id) return Response.json({ error: "Unauthorized" }, { status: 401 });
  const userId = Number(session.user.id);

  const { transcript } = await req.json();
  if (!transcript?.trim()) {
    return Response.json({ error: "Transcript vacío" }, { status: 400 });
  }

  // 1. Parsear con Gemini
  const model = genAI.getGenerativeModel({ model: "gemini-2.5-flash" });
  const controller = new AbortController();
  setTimeout(() => controller.abort(), 6000);

  try {
    const result = await model.generateContent([
      { text: VOICE_PARSE_PROMPT },
      { text: `\n\nTranscript: "${transcript}"` },
    ]);
    const text = result.response.text();
    const cleaned = text.replace(/```json|```/g, "").trim();
    const parsed = JSON.parse(cleaned);

    // 2. Matchear cada item contra DB del usuario (respeta CO regional)
    const matched = await matchFoods(parsed.items, userId);

    return Response.json({ items: matched, transcript });
  } catch (e) {
    console.error("[parse-voice]", e);
    return Response.json({ error: "Error al procesar" }, { status: 500 });
  }
}
```

### `lib/voice/match-foods.ts`

```ts
import { prisma } from "@/lib/prisma";

type ParsedItem = {
  nombre: string;
  cantidad: number;
  unidad: string;
  confianza: number;
};

type MatchedItem = ParsedItem & {
  foodId: number | null;
  cal?: number; p?: number; c?: number; f?: number;
  weightG: number;
};

const UNIT_TO_GRAMS: Record<string, number> = {
  taza: 240,
  cucharada: 15,
  porcion: 150,
  unidad: -1, // depende del food (gramsPerUnit)
};

export async function matchFoods(items: ParsedItem[], userId: number): Promise<MatchedItem[]> {
  const user = await prisma.user.findUnique({ where: { id: userId }, select: { countryCode: true } });
  const matched: MatchedItem[] = [];

  for (const item of items) {
    // Búsqueda priorizada por regionCode = user.countryCode
    const food = await prisma.food.findFirst({
      where: {
        AND: [
          { nombre: { contains: item.nombre.toLowerCase() } },
          { OR: [{ userId }, { userId: null }] },
        ],
      },
      orderBy: [
        { regionCode: user?.countryCode ? "desc" : "asc" },
        { verifiedAt: { sort: "desc", nulls: "last" } },
        { usageCount: "desc" },
      ],
    });

    let weightG = 0;
    if (item.unidad === "g") weightG = item.cantidad;
    else if (item.unidad === "ml") weightG = item.cantidad; // aprox para densidad ~1
    else if (item.unidad === "unidad" && food?.gramsPerUnit) {
      weightG = item.cantidad * food.gramsPerUnit;
    } else {
      weightG = item.cantidad * (UNIT_TO_GRAMS[item.unidad] ?? 100);
    }

    const scale = weightG / 100;
    matched.push({
      ...item,
      foodId: food?.id ?? null,
      cal: food ? food.cal * scale : undefined,
      p: food ? food.p * scale : undefined,
      c: food ? food.c * scale : undefined,
      f: food ? food.f * scale : undefined,
      weightG,
    });
  }

  return matched;
}
```

```ts
import { matchFoods } from "@/lib/voice/match-foods";

describe("matchFoods", () => {
  it("matches 'arepa' to Colombian regional food when user is CO", async () => {
    // Setup: usuario CO + Food "Arepa" con regionCode="CO"
    const items = [{ nombre: "arepa", cantidad: 1, unidad: "unidad", confianza: 0.9 }];
    const result = await matchFoods(items, userIdCO);
    expect(result[0].foodId).toBeDefined();
    expect(result[0].weightG).toBeGreaterThan(0);

  it("converts 'taza' to 240g", async () => {
    const items = [{ nombre: "leche", cantidad: 1, unidad: "taza", confianza: 0.9 }];
```
