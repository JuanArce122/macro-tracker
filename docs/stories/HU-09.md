## Objetivo
Generar insights semanales personalizados basados en patrones reales del usuario (tendencia de peso, adherencia, racha, mejoras) con un máximo de 2 notificaciones push por semana.

## Motor de generación

### `lib/insights/rules.ts`

```ts
import { prisma } from "@/lib/prisma";
import { calcTrendSlope, calcAdherence } from "@/lib/weight-trend";

type GeneratedInsight = {
  type: string;
  title: string;
  body: string;
  data: Record<string, unknown>;
};

/**
 * RULE 1: Tendencia de calorías vs meta (últimos 14 días)
 */
export async function ruleCaloriesTrend(userId: number): Promise<GeneratedInsight | null> {
  const fourteen = new Date();
  fourteen.setDate(fourteen.getDate() - 14);
  const meals = await prisma.meal.findMany({
    where: { userId, date: { gte: fourteen } },
    select: { dateLocal: true, calories: true },
  });
  const goal = await prisma.goal.findFirst({ where: { userId } });
  if (!goal || meals.length < 7) return null;

  const byDay = new Map<string, number>();
  meals.forEach((m) => {
    const day = m.dateLocal ?? "";
    byDay.set(day, (byDay.get(day) ?? 0) + m.calories);
  });
  const avgCal = Array.from(byDay.values()).reduce((s, c) => s + c, 0) / byDay.size;
  const diff = avgCal - goal.calories;
  const pctDiff = (diff / goal.calories) * 100;

  if (Math.abs(pctDiff) < 10) return null;

  return {
    type: "trend_calories",
    title: pctDiff > 0 ? "Estás comiendo por encima de tu meta" : "Estás comiendo por debajo de tu meta",
    body: `Tu promedio de 14 días es ${avgCal.toFixed(0)} kcal (meta ${goal.calories}). ${
      pctDiff > 0 ? "Considera ajustar porciones." : "Asegúrate de comer suficiente."
    }`,
    data: { avgCal, targetCal: goal.calories, pctDiff },
  };
}

/**
 * RULE 2: Adherencia baja (<60% en 14 días)
 */
export async function ruleAdherenceLow(userId: number): Promise<GeneratedInsight | null> {
  const adherence = await calcAdherence(userId);
  if (adherence >= 0.6) return null;

  return {
    type: "adherence",
    title: "Llevas días sin registrar",
    body: `Has registrado el ${(adherence * 100).toFixed(0)}% de los últimos 14 días. Activa recordatorios para mantener consistencia.`,
    data: { adherence },
  };
}

/**
 * RULE 3: Racha (7+ días consecutivos)
 */
export async function ruleStreak(userId: number): Promise<GeneratedInsight | null> {
  // Calcular racha actual
  const meals = await prisma.meal.findMany({
    where: { userId },
    select: { dateLocal: true },
    orderBy: { date: "desc" },
  });
  const distinctDates = Array.from(new Set(meals.map((m) => m.dateLocal))).filter(Boolean).sort().reverse();
  let streak = 0;
  let prevDate = new Date();
  for (const d of distinctDates) {
    const date = new Date(d!);
    const diffDays = Math.round((prevDate.getTime() - date.getTime()) / 86400000);
    if (diffDays <= 1) { streak++; prevDate = date; } else break;
  }

  if (streak < 7) return null;
  // Solo notificar cuando alcanza hitos: 7, 14, 30, 60, 90
  if (![7, 14, 30, 60, 90].includes(streak)) return null;

  return {
    type: "streak",
    title: `${streak} días seguidos registrando`,
    body: streak === 7 ? "¡Buena racha! El tracking consistente es la clave." : "Sigue así.",
    data: { streak },
  };
}

/**
 * RULE 4: Desbalance de macros
 */
export async function ruleMacroImbalance(userId: number): Promise<GeneratedInsight | null> {
  // Si proteína promedio < 70% del goal 7 días seguidos
  const goal = await prisma.goal.findFirst({ where: { userId } });
  if (!goal) return null;

  const seven = new Date();
  seven.setDate(seven.getDate() - 7);
  const meals = await prisma.meal.findMany({
    where: { userId, date: { gte: seven } },
    select: { dateLocal: true, protein: true },
  });
  const byDay = new Map<string, number>();
  meals.forEach((m) => {
    const day = m.dateLocal ?? "";
    byDay.set(day, (byDay.get(day) ?? 0) + m.protein);
  });
  const avgProt = Array.from(byDay.values()).reduce((s, p) => s + p, 0) / byDay.size;
  const pctOfGoal = (avgProt / goal.protein) * 100;

  if (pctOfGoal >= 70) return null;

  return {
    type: "macro_imbalance",
    title: "Proteína por debajo de tu meta",
    body: `Tu promedio es ${avgProt.toFixed(0)}g (meta ${goal.protein}g). Considera huevos, lácteos, legumbres o pollo.`,
    data: { avgProtein: avgProt, goalProtein: goal.protein },
  };
}
```

### `lib/insights/generator.ts`

```ts
export async function generateInsightsForUser(userId: number): Promise<void> {
  // No generar si tiene >2 insights activos esta semana
  const weekAgo = new Date();
  weekAgo.setDate(weekAgo.getDate() - 7);
  const recent = await prisma.insight.count({
    where: { userId, createdAt: { gte: weekAgo }, pushedAt: { not: null } },
  });
  if (recent >= 2) return;

  // Evaluar todas las reglas y elegir la 1-2 más relevantes
  const rules = [ruleCaloriesTrend, ruleAdherenceLow, ruleStreak, ruleMacroImbalance];
  const generated: GeneratedInsight[] = [];
  for (const rule of rules) {
    try {
      const result = await rule(userId);
      if (result) generated.push(result);
    } catch (e) { console.error(`[insight rule] ${e}`); }
  }

  // Limitar a 1 por semana (2 si hay milestone)
  const milestones = generated.filter((g) => g.type === "streak");
  const others = generated.filter((g) => g.type !== "streak").slice(0, 1);
  const finalList = [...milestones, ...others].slice(0, 2);

  for (const insight of finalList) {
    const saved = await prisma.insight.create({
      data: {
        userId,
        type: insight.type,
        title: insight.title,
        body: insight.body,
        dataJson: JSON.stringify(insight.data),
      },
    });
    await sendPushToUser(userId, saved);
  }
}
```

## Endpoint cron

### `app/api/cron/insights/route.ts`

```ts
import { auth } from "@/auth";
import { generateInsightsForUser } from "@/lib/insights/generator";
import { prisma } from "@/lib/prisma";

export async function GET(req: Request) {
  // Vercel envía Authorization: Bearer <CRON_SECRET>
  const authHeader = req.headers.get("authorization");
  if (authHeader !== `Bearer ${process.env.CRON_SECRET}`) {
    return Response.json({ error: "Unauthorized" }, { status: 401 });
  }

  // Solo procesar usuarios con ≥7 días de tracking
  const users = await prisma.user.findMany({
    where: {
      meals: { some: { createdAt: { gte: new Date(Date.now() - 14 * 24 * 60 * 60 * 1000) } } },
    },
    select: { id: true },
  });

  for (const user of users) {
    await generateInsightsForUser(user.id);
  }

  return Response.json({ ok: true, processed: users.length });
}
```

## Web Push

### `lib/push.ts`

```ts
import webpush from "web-push";

webpush.setVapidDetails(
  process.env.VAPID_SUBJECT!,
  process.env.VAPID_PUBLIC_KEY!,
  process.env.VAPID_PRIVATE_KEY!
);

export async function sendPushToUser(userId: number, insight: Insight) {
  const subs = await prisma.pushSubscription.findMany({ where: { userId } });
  const payload = JSON.stringify({
    title: insight.title,
    body: insight.body,
    url: "/insights",
    insightId: insight.id,
  });

  for (const sub of subs) {
    try {
      await webpush.sendNotification({
        endpoint: sub.endpoint,
        keys: { p256dh: sub.p256dh, auth: sub.auth },
      }, payload);
    } catch (err) {
      if ((err as { statusCode: number }).statusCode === 410) {
        // Subscription expired
        await prisma.pushSubscription.delete({ where: { id: sub.id } });
      }
    }
  }

  await prisma.insight.update({
    where: { id: insight.id },
    data: { pushedAt: new Date() },
  });
}
```

## Riesgos específicos

| Riesgo | Mitigación |
|---|---|
| Web Push no funciona en iOS hasta 16.4+ y solo PWA instalada | Documentar en `/settings/notifications`; fallback in-app card |
| Insights spam si el cron corre 2 veces | `pushedAt` previene duplicados; CRON_SECRET protege |
| Algoritmos pueden generar mensajes culpabilizadores | Tono empático en `body`; QA manual de todos los templates |
| Vercel cron tiene timeout de 60s en plan Pro | Paginar usuarios si > 1000; procesar en batches |
